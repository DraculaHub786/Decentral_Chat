  151:         async function sendMessage() {
  152:             const input = document.getElementById('messageInput');
  153:             const content = input.value.trim();
  154: 
  155:             if (!content && selectedFiles.length === 0) return;
  156:             if (!activeChat) {
  157:                 showError('No active chat selected');
  158:                 return;
  159:             }
  160: 
  161:             const sendBtn = document.getElementById('sendBtn');
  162:             sendBtn.disabled = true;
  163: 
  164:             try {
  165:                 let attachmentIds = [];
  166:                 let attachmentTypes = [];  // ✅ Store file types
  167: 
  168:                 // Upload files if any
  169:                 if (selectedFiles.length > 0) {
  170:                     console.log(`📤 Uploading ${selectedFiles.length} files...`);
  171: 
  172:                     // Show uploading indicator
  173:                     showNotification('Uploading', `Uploading ${selectedFiles.length} file(s)...`);
  174: 
  175:                     try {
  176:                         const uploadResult = await uploadFiles();
  177:                         attachmentIds = uploadResult.ids || [];
  178:                         attachmentTypes = uploadResult.types || [];  // ✅ Get file types
  179: 
  180:                         if (attachmentIds.length === 0) {
  181:                             throw new Error('No files were uploaded successfully');
  182:                         }
  183: 
  184:                         // Verify we got valid IDs
  185:                         if (attachmentIds.some(id => !id || id === 'undefined')) {
  186:                             throw new Error('Invalid file IDs received from server');
  187:                         }
  188: 
  189:                         console.log('✅ Files uploaded successfully, IDs:', attachmentIds, 'Types:', attachmentTypes);
  190: 
  191:                         // Clear file selection
  192:                         selectedFiles = [];
  193:                         const filePreviewItems = document.getElementById('filePreviewItems');
  194:                         if (filePreviewItems) {
  195:                             filePreviewItems.innerHTML = '';
  196:                         }
  197:                         document.getElementById('filePreviewContainer').style.display = 'none';
  198: 
  199:                         showNotification('Success', `${attachmentIds.length} file(s) uploaded`);
  200: 
  201:                     } catch (uploadError) {
  202:                         console.error('❌ File upload failed:', uploadError);
  203:                         showError('Failed to upload files: ' + uploadError.message);
  204:                         sendBtn.disabled = false;
  205:                         return; // Stop here if upload fails
  206:                     }
  207:                 }
  208: 
  209:                 // Send message via WebSocket
  210:                 if (ws && ws.readyState === WebSocket.OPEN) {
  211:                     // 🔒 Encrypt message content if E2E key is available
  212:                     let plainContent = content || '';
  213:                     let encryptedContent = plainContent;
  214:                     let isE2eEncrypted = false;
  215: 
  216:                     if (plainContent && CryptoEngine.hasKey(activeChat.id)) {
  217:                         const ct = await CryptoEngine.encryptText(activeChat.id, plainContent);
  218:                         if (ct) {
  219:                             encryptedContent = ct;
  220:                             isE2eEncrypted = true;
  221:                         }
  222:                     }
  223: 
  224:                     // If any attached file was encrypted, mark the whole message encrypted
  225:                     if (!isE2eEncrypted && attachmentIds.length > 0 && window._encryptedFileIds) {
  226:                         isE2eEncrypted = attachmentIds.some(id => window._encryptedFileIds.has(id));
  227:                     }
  228: 
  229:                     // ✅ Determine message type based on uploaded file types
  230:                     let messageType = 'text';
  231:                     if (attachmentIds.length > 0) {
  232:                         // Use first file's type, or 'file' if mixed types
  233:                         const firstType = attachmentTypes[0] || 'file';
  234:                         const allSameType = attachmentTypes.every(t => t === firstType);
  235:                         messageType = allSameType ? firstType : 'file';
  236:                     }
  237: 
  238:                     const messageData = {
  239:                         type: 'message',
  240:                         chat_id: activeChat.id,
  241:                         content: encryptedContent,
  242:                         encrypted: isE2eEncrypted,
  243:                         message_type: messageType,  // ✅ Use detected type
  244:                         attachments: attachmentIds,
  245:                         reply_to: replyToMessageId || null
  246:                     };
  247: 
  248:                     console.log('📤 Sending message via WebSocket:', messageData);
  249:                     ws.send(JSON.stringify(messageData));
  250: 
  251:                     // Clear input
  252:                     input.value = '';
  253:                     input.style.height = 'auto';
  254: 
  255:                     // Clear reply indicator
  256:                     if (replyToMessageId) {
  257:                         cancelReply();
  258:                     }
  259: 
  260:                     // Stop typing indicator
  261:                     if (isTyping) {
  262:                         isTyping = false;
  263:                         ws.send(JSON.stringify({
  264:                             type: 'typing',
  265:                             chat_id: activeChat.id,
  266:                             is_typing: false
  267:                         }));
  268:                     }
  269: 
  270:                     console.log('✅ Message sent successfully');
